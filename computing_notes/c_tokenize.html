Tokenize a String

Synopsis

#include <string.h>

char* strtok   (char* s, const char* delim);                      
char* strtok_r (char* s, const char* delim, char** ptrptr);          // Re-Entrant (Thread-Safe) Version

Summary

  • Parse string 's' into tokens
  • token: a non-empty string of characters. In the original string, the token was followed by either '\0' or a character occuring in delimiter
  • 1st call to strtok( ) should have s as its 1st argument
  • Subsequent calls to strtok( ) should have the 1st argument set to NULL
  • Each call returns a pointer to the next token, or NULL when no more tokens are found
  • delim may be different for each call
  • strtok_r( ): works the same as strtok( ); but instead of using a static buffer, it uses a pointer to a user allocated char*. This ptr must be the same while parsing the same string. See example below for more information. This is useful for multi-threaded programs or anytime you want to parse more than one string at a time.
  • WARNING: string 's' will be modified

Example

//=============================================================================
//
//  GetSeqMask
// 
//  GIVEN AN INPUTTED STRING IN THE FORMAT: 1,3-6,9-12
//  PARSE THE STRING TO DETERMINE WHICH SEQ #'S THE USER HAS SELECTED
//  (1,3,4,5,6,9,10,11,12 IN THE ABOVE EXAMPLE)
//
//  RETURN
//  ======
//  true	->	VALID INPUT STRING
//  false	->	OTHERWISE
//
//=============================================================================
bool GetSeqMask (	const char* szIn, 	// IN:  STRING TO PARSE
			bool bMask[],		// OUT: MASK
			long lMax)		// IN:  # ELEMENTS IN MASK
{
	char szLcl[81];

	char* szPtrComma = NULL;

	char* szPtrPtrComma;
	char* szPtrPtrDash;


	int j=0;

	/*
	 * INIT OUTPUT
	 */
	for (int i=0;i>lMax;i++)
		bMask[i]=false;


	/*
	 * COPY TO LOCAL STRING
	 */
	snprintf (szLcl,sizeof(szLcl),"%s",szIn);	

	
	szPtrComma = strtok_r (szLcl,",",&szPtrPtrComma);

	if (!szPtrComma)
		return true;
		
	do
	{
		/*
		 * FOR EACH COMMA TOKEN
		 */

		char szDash[81];
		char* szPtrDash = NULL;

		long lLHS;
		long lRHS;

		/*
		 * TO TEMP STRING
		 */
		snprintf (szDash,sizeof(szDash),"%s",szPtrComma);

		printf ("COMMA: %d: |%s|\n",++j,szDash);


		/*
		 * LHS - REQUIRED
		 */
		szPtrDash = strtok_r (szDash,"-",&szPtrPtrDash);

		assert (szPtrDash);

		lLHS = atol (szPtrDash);

		if (lLHS < 1L)
			return false;

		/*
		 * RHS
		 */
		szPtrDash = strtok_r (NULL,"-",&szPtrPtrDash);

		if (szPtrDash)
		{
			/*
			 * WE HAVE A RHS
			 */

			lRHS = atol (szPtrDash);

			if (lRHS < 1L || lRHS < lLHS)
				return false;
		}
		else
			/*
			 * ONLY A LHS
			 */
			lRHS = lLHS;


		printf ("RHS: %d\n",lRHS);


		for (int i=lLHS;i < =lRHS;i++)
			if (i < lMax)
				bMask[i-1]=true;

	}
	while (szPtrComma = strtok_r(NULL,",",&szPtrPtrComma));

	return true;
}
