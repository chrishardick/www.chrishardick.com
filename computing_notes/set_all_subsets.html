Set: All Subsets

  • Given a set of n items, return all possible subsets including the empty set
  • There are 2n sub-sets 
GIVEN: s = { 'a, 'b' }

FIND: All possible subsets, including the empty set 

ANSWER:
{ { }, {'a'}, {'b'}, {'a','b'} }

Recursive Solution

#include <iostream>
#include <string>

#include <ch_vector.h>

using namespace std;

// given a set of items, return a list of all subsets - including the empty sub-set


/*

walk thru and test...

all_subsets ([a, b])
    |
    first = a
    rem = [b]
    |
    vector<vector<char>> setsWithoutFirst = all_subsets (rem [b])
        |
        first = b
        rem = []
        |
        vector<vector<char>> setsWithoutFirst   = all_subsets (rem [])
                                                = [[]]
        |
        vector<vector<char>> setsWithFirst = []
        |
        traverse setsWithoutFirst // [[]]
            |
            vector<char> tmp
            tmp.push_back(first)    // b
            tmp.insert(tmp.end(), *a.begin(), *a.end())
            |
            setsWithoutFirst.push_back(tmp);
        |
        return (setsWithFirst + setsWithoutFirst);  // [ [b], [] ]
    |
    vector<vector<char>> setsWithFirst = []
    |
    traverse setsWithoutFirst   // [ [b], [] ]
        |
        for each set, append first
        append the set to setsWithFirst
        results in [ [a,b], [a] ]
    |
    return (setsWithFirst + setsWithoutFirst)   // [a,b], [a], [b], []
*/

vector<vector<char>> all_subsets (const vector<char>& v)
{
    if (v.size() == 0)
    {
        return vector<vector<char>>(1);
    }

    char first = *(v.begin());
    vector<char> rem(v.begin()+1, v.end());

    vector<vector<char>> setsWithoutFirst = all_subsets(rem);
    vector<vector<char>> setsWithFirst = vector<vector<char>>();

    // for each set without first, add the first (setsWithFirst)
    for (auto a = setsWithoutFirst.begin(); a != setsWithoutFirst.end(); a++)
    {
        vector<char>& aa = *a;

        vector<char> tmp;

        tmp.push_back (first);
        tmp.insert (tmp.end(), aa.begin(), aa.end());

        setsWithFirst.push_back(tmp);
    }

    // return the set union
    setsWithFirst.insert (setsWithFirst.end(), setsWithoutFirst.begin(), setsWithoutFirst.end());
    return setsWithFirst;
}


int main (int argc, const char* argv[])
{
    (void)argc;
    (void)argv;

    vector<char> v { 'a', 'b' };

    vector<vector<char>> sets = all_subsets(v);

    for (auto a=sets.begin(); a!=sets.end(); a++)
        cout << *a << endl;
        
    return 0;
}
