C++ Unordered Set, Unordered Multiset

  • Set of objects of type Key with constant time addition, search and removal.
  • Sets have unique elements.
  • Multisets allows duplicates
  • See also C++ Set
  • See also C++ Unordered Map and Unordered Multimap
  • Requires C++ 11 
#include <unordered_set>

template < class Key
        ,class Hash = std::hash<Key>
        ,class KeyEqual = std::equal_to<Key>
        ,class Allocator = std::allocator<Key>
        > class unordered_set;

template < class Key
        ,class Hash = std::hash<Key>
        ,class KeyEqual = std::equal_to<Key>
        ,class Allocator = std::allocator<Key>
        > class unordered_multiset;
