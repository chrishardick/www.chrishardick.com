C++ - std::inserter and std::back_inserter

#include <iterator>

//
// constructs an input iterator which inserts new elements into Container x
//
template <typename Container>
insert_iterator<Container> inserter (Containter& x, Container::iterator it);

//
// constructs a back-insert-iterator which inserts new elements at the end of x
//
template <typename Container>
back_insert_iterator<Container> back_inserter (Container& x);
