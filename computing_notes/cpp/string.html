The Standard C++ String Class

  • C++ Header File: string
  • C++ string class is in the std namespace
  • A simple example 
    #include <iostream>
#include <string>

using namespace std;

int main (int argc, char* argv[])
{
        string s1 = "Hello";

        string s2 ("World");

        string s3 = s1 + " " + s2;

        cout << s3 << endl;

        return 0;
}
