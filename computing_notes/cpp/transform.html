std::transform

  • Transform a sequence of elements by calling unary operator op on each element and place the result in result
  • There is also a binary version which accepts 2 sets of InputIterators
  • The operator can be either a function or a functor 
    #include <algorithm>

// unary  
template <class InputIterator, class OutputIterator, class UnaryOperation>
OutputIterator std::transform ( InputIterator begin
                                ,InputIterator end
                                ,OutputIterator result
                                ,UnaryOperation op
                                );

#include <vector>:1

vector<int> a;
vector<int> b;

// pof (plain old function)
int add1 (int x)    { return x+1 }

// functor (function object)
class AddX 
{
    private:
        int d_x;

    public:
        AddX (int x) : d_x(x) { }

        int operator() (int y) { return d_x + y; }
};

for (int i=0;i<5;i++)
{
    a.push_back(i);
}

b.resize(a.size);


// function example

std::transform (a.begin(),a.end(),b.begin(),add1);

// functor example
// - w/o functors, you'd have to write 3 functions or use a function template
AddX add_5(5);
AddX add_10(10);
AddX add_20(20);

std::transform (a.begin(),a.end(),b.begin(),add_5);
