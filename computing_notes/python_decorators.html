Python: Decorators

  • A decorator is a function which accepts a function and extends its behavior
  • In other words, decorators wrap a function
  • An example: 
def before_after_decorator (func):
    def wrapper():
        print ("before")
        func()
        print ("after")

    return wrapper

def during():
    print ("during")


# note that we're replacing the original function
during = before_after_decorator(during)

during()

# output:
before
during
after

Pie Synatax

# Equivalent to: 
# during = before_after_decorator(during)

@before_after_decorator                     
def during():
    print ("during")

during()

# output:
before
during
after

Decorators with Arguments

  • Add *args, **kwargs to inner function such that it can accept arguments 
def before_after_decorator(func):
    def wrapper (*args,**kwargs):
        print ("before")
        func (*args,**kwargs)
        print ("after")
    return wrapper

def foo(str):
    print (str)

foo = before_after_decorator (foo)

# equivalent - using pie syntax
@before_after_decorator
def foo (str):
    print (str)


foo ("Hello, World!")

# output
before
Hello, World!
after

Placing Decorators in Modules

  • Normally, decorators are placed in modules such that they can be re-used. 
# do_twice.py

def do_twice(func):
    def wrapper ():
        func()
        func()
    return wrapper


# hello_world.py

from do_twice import do_twice

@do_twice
def hello_world():
    print ("Hello, World!")


hello_world()

# output
Hello, World!
Hello, World!

Timer

import functools
import datetime
import time

def timer (func):
    def wrapper(*args,**kwargs):
        t1 = datetime.datetime.utcnow()
        func(*args, **kwargs)
        t2 = datetime.datetime.utcnow()
        print ("%s: Elapsed time: %s sec" % (func.__name__, t2-t1)
    return wrapper

@timer
def my_sleep (n):

    print ("my_sleep: Sleeping %d..." % (n)
    time.sleep(n)


my_sleep (10)

# output
my_sleep: Sleeping 10...
my_sleep: Elapsed time: 10.001 sec
