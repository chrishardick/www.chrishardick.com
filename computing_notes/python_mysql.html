Create Database

import mysql.connector

host    = "localhost"
user    = "bhah"
passwd  = "blah"

db_conn = mysql.connector.connect (host=host, user=user, passwd=passwd)

#
# all DDL (Data Definition Language) statements are executed using a cursor (a handle data structure)
# - DDL statements implicitly xecute commit 
# - a DDL statement is a statement which alters the structure of the database itself 
# - e.g. CREATE, ALTER, DROP
#

db_cursor = db_conn.cursor()

db_cursor.execute ("CREATE DATABASE foo")

db_cursor.execute ("USE foo")

db_cursor.execute ("CREATE TABLE bar (bar1 VARCHAR(255), bar2 VARCHAR(255))")




db_cursor.execute ("CREATE DATABASE blah")

db_cursor.execute ("USE blah")
