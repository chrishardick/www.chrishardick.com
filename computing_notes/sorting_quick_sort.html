Quick Sort

  • Recursive Algorithm
  • A Divide and Conquor Algorithm

Complexity

  • O(n log(n))

Algorithm

// try this example: [ 9 3 5 2 6 ]

void print (std::ostream& os, std::vector<int>& v)
{
    for (auto a = v.begin(); a != v.end(); a++)
    {
        os << *a << " ";
    }

    os << std::endl;
}

void swap (int& a, int& b)
{
    int c = a;

    a = b;
    b = c;
}

int partition (std::vector<int>& v, int start_idx, int end_idx)
{
    std::cout << "partition: start=" << start_idx << " end=" << end_idx << std::endl;

    int pivot = v[end_idx];

    int i = start_idx - 1;

    for (int j = start_idx; j <= end_idx - 1; j++)
    {
        if (v[j] < pivot)
        {
            i++;
            swap (v[i], v[j]);
        }
    }

    swap(v[i+1], v[end_idx]);

    print (std::cout, v);

    return i+1;
}

void quick_sort (std::vector& v, int start_idx, int end_idx)
{
    if (start_idx >= end_idx)
        return;

    int p_idx = partition (v, start_idx, end_idx);

    quick_sort (v, start_idx, p_idx - 1);     // before
    quick_sort (v, p_idx +1, end_idx);        // after
}
