CREATE TABLE

    CREATE TABLE data (src varchar(50),dest varchar(50),proto varchar(10),num_pkts int, num_bytes int, timestamp varchar(25))

CREATE INDEX

    CREATE [UNIQUE] INDEX index-name ON table-name (indexed-column [ASC|DESC], indexed-column [ASC|DESC], ...)
  • Example(s): 
    CREATE UNIQUE INDEX src_dest_proto_timestamp ON data (src, dest, proto, timestamp)
