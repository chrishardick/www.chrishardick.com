Node.js - require

  • require - a builtin function which allows you to include modules which are defined in a different file
  • require will read a JavaScript file, execute it and return the exports object 
// ch.js
// - useful js utilities

module.exports = {
    foo: function 
    { 
        console.log ("foo");        
    },
    bar: function 
    {
        console.log ("bar");
    }
};

//
// app.js
//
ch = require ('./ch');

ch.foo();
ch.bar();
