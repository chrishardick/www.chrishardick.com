C Function Reference

assert ()

//
// Abort program printing an informative message to stdio if expression is false
// - implemented using a macro
// - assert() is disabled if macro NDEBUG has been defined 
// - NDEBUG is usally defined for production builds
// - This is usually set by the makefile generator (E.g. CMake) (-DNDEBUG)
//
#include <assert.h>

void assert (expression);

fgets ()

/*
 * READ @ most size-1 characters from stream and store in s
 * reading stops after EOF or \n
 * if \n is read, it is stored in the buffer
 * returned string is always null (\0) terminated
 * 
 * RETURN
 * ======
 * s = success
 * NulL = otherwise
 */

#include <stdio.h>

char* fgets (char* s,        // out:    buffer to write to (usually of size 'size')
             int size,       // in:     size of buffer 
             FILE* stream);  // in/out: stream to read from
