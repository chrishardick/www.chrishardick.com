Merge Sort

  • Recursive Algorithm
  • A Divide and Conquor Algorithm
  • Divides the input array into two halves
  • Calls itself for each half
  • Merges the two halves
  • Works on both lists and arrays

Complexity

  • O(n log(n))

Algorithm

// try this example: [ 6 2 5 3 9 ]

//
// merge two sorted subarrays of v
// [start_idx] - [mid_idx]
// [mid_idx+1] - [end_idx]
//
// put the results back in v
//
void merge (std::vector<int>& v
            ,int start_idx
            ,int mid_idx
            ,int end_idx
            )
{
    int size1 = mid_idx - start_idx + 1;
    int size2 = end_idx - mid_idx;

    std::cout   << "merge:"
                << " start=" << start_idx << " mid=" << mid_idx << " end=" << end_idx
                << " size1=" << size1 << " size2=" << size2 << std::endl;

    // temp vectors
    std::vector<int> v1(&v[start_idx], &v[mid_idx+1]);      // idx = i
    std::vector<int> v2(&v[mid_idx+1], &v[end_idx+1]);      // idx = j

    std::cout << "v1: ";
    print(std::cout,v1);
    
    std::cout << "v2: ";
    print(std::cout,v2);

    int i=0, j=0, k=start_idx;

    while (i < size1 && j < size2)
    {
        if (v1[i] < v2[j])
        {
            v[k++] = v1[i++];
        }
        else
        {
            v[k++] = v2[j++];
        }
    }

    // copy any remaining

    while (i < size1)
    {
        v[k++] = v1[i++];
    }

    while (j < size2)
    {
        v[k++] = v2[j++];
    }
}

void merge_sort (std::vector<int>& v
                ,int start_idx          // e.g. 0
                ,int end_idx            // e.g. size-1
                )
{
    std::cout << "merge_sort: start: " << start_idx << " end: " << end_idx << std::endl;

    if (start_idx >= end_idx)
    {
        std::cout << "merge_sort: no action" << std::endl;
        return;
    }

    int mid = (start_idx + end_idx) / 2;

    merge_sort (v, start_idx, mid);

    merge_sort (v, mid+1, end_idx);

    merge (v, start_idx, mid, end_idx);
}
