Python: HTTP Cookies

Cookie - HTTP State Management Library

Setting Cookies

import Cookie

c = Cookie.simpleCookie()

#
# name and value
#
c['session_data']    = session_id

# expires
# - optional. defaults to a transient cookie (lasts until the user closes the browser)
# - specifying a value causes the brower to store the cookie in a local file
# - once the cookie expires, the browser should no longer communicate the cookie to the server and the file may be deleted
c['session_id']['expires'] = 30*60          # 30 minutes

# path
# - optional. defaults to the web page which created the cookie ad any other web page in the same directory or sub directory
c['session_id']['path']    = 30*60          

# headers - start
print c                             
print "Content-type: text/html"

# headers end (headers always end with a blank line)
print ""                
print "<html>"
print "..."
print "</html>"

Retrieving Cookies

import os
import Cookie

if 'HTTP_COOKIE' in os.environ:
    cookie_string = os.environ.get('HTTP_COOKIE')
    c = Cookie.SimpleCookie()
    c.load(cookie_string)

    try:

        data = c['session_id'].value

    except KeyError:

        print "Cookie Not Found or Expired"

Deleting Cookies

Cookies are usually deleted by setting an expiration time in the past

import Cookie
c = Cookie.SimpleCookie()
c['session_id']=''
c['session_id']['expires']='Thu 01 Jan 1970 00:00:00 GMT'

print c
print "Content-type: text/html"
print ""
