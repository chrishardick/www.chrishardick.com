Graph - Breadth First

  • Breadth First always involves a queue
  • You can use a list in C++, collections.deque in Python, etc.

Basic Algorithm

  • Put one of the graph's nodes in the queue
  • while the queue is not empty:
    • Remove the front item from the queue and add it to the visited set
    • For each adjacent node which has yet to be visited, add it to the queue

Example

struct Point
{
    int x {0};
    int y {0};

    int num {0};

    Point (int _x=0, int _y=0, int _num=0) :
        x(_x), y(_y), num(_num)
        {}
};


int num_steps (const vector<vector<char>>& board, int x, int y)
{
    set<pair<int,int>> visited;

    list<Point> q;

    q.push_back(Point(x,y));

    while (!q.empty())
    {
        auto p = q.front();

        q.pop_front();

        visited.insert(make_pair(p.x,p.y));

        if (board[p.y][p.x] == 'E')
            return p.num;

        vector<Point> pts { Point(p.x+1,p.y,p.num+1)
                            ,Point(p.x-1,p.y,p.num+1)
                            ,Point(p.x,p.y+1,p.num+1)
                            ,Point(p.x,p.y-1,p.num+1)
                            };
       
        for (auto a = pts.begin(); a != pts.end(); a++)
        {
            Point& pt_ref = *a;

            if (pt_ref.x >= 0 && pt_ref.x <= (int)board[0].size()-1 && 
                pt_ref.y >= 0 && pt_ref.y <= (int)board.size()-1)
            {
                if (visited.find(make_pair(pt_ref.x,pt_ref.y)) == visited.end())
                    q.push_back(pt_ref);
            }
        }
    }

    return 0;
}

int main (int argc, const char* argv[])
{
    (void)argc;
    (void)argv;

    int steps = 0;

    vector<vector<char>> board {
            //   0    1    2    3    4
             {  ' ', ' ', ' ', ' ', ' ' }
            ,{  'S', ' ', ' ', ' ', ' ' }
            ,{  ' ', ' ', ' ', ' ', ' ' }
            ,{  ' ', ' ', 'E', ' ', ' ' }
            };

    for (unsigned int y=0; y<board.size(); y++)
    {
        for (unsigned int x=0; x<board[0].size(); x++)
        {
            if (board[y][x] == 'S')
            {
                steps = num_steps (board, x, y);
                break;
            }
        }
    }

    cout << "#steps=" << steps << endl;

    return 0;
}
