C/UNIX: getopt()

  • process command line options 
#include <unistd.h>

extern char* optarg;
extern int optind, opterr, optopt;

//
// optstring - a string containing valid options letters
//  if any letter is followed by ':', it requires an option
//  if any letter is followed by '::', it has an optional argument in same argv-element (e.g. -oarg) - GNU extension
//
// getopt() can detect two types of errors:
// (1) an invalid option (not specified in optstring)
// (2) a known option requiring an argument without an argument
// by default, getopt() will print an error message to standard error and retunn '?' as the return value
//

int getopt (int argc
            ,char* const argv[]
            ,const char* optstring
            );


//
// typical usage
//

int c = 0;

bool a_option = false;

std::string b_option_value;

while((c = getopt(argc, argv,"ab:h")) != -1)
{
    switch (c)
    {
        case 'a':
            // process option a

            a_option = true;

            break;

        case 'b':
            // process option b

            b_option_value = optarg;        // optarg contains the passed in argument 

            break;

        case 'h': // usage

            usage();

            return -1;
        
        default:
            
            std::cout << std::endl;

            usage();

            return -1;
    }
}
