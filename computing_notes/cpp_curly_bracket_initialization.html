C++ - Curly Bracket Initialization

int arr[] = { 1, 2, 3, 4 };

std::vector<int> v { 1, 2, 3, 4};

std::map<std::string, int> m { {"aa", 1}, {"bb", 2}, {"cc", 3} };


//
// initialize with default values
//

int i { };                  // i = 0
double d { };               // d = 0.0

std::string s { }           // s = ""

std::vector v { }   // empty vector


//
// struct
//

struct S
{
    int x;
    double y;
};

S s { 1, 1.1 };


//
// class
//

class C
{
    int x;
    double y;

    public:
        C (int xx, int yy) : x(xx), y(yy) { }
};


C c { 2, 2.2 };
