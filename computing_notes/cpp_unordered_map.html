C++ Unordered Map

  • I.e. Hash Table
  • key value - essentially the key (lhs)
  • mapped value - essentially the value (rhs)
  • value_type - a pair class in which the first value is a const version of the key value and the second value is the mapped value 
#include <unordered_map>

template < class Key
                ,class T
                ,class Hash = hash<Key>
                ,class Pred = equal_to<Key>
                ,class Alloc = allocator< pair<const Key, T> >
                > class unordered_map;
