Binary Tree - BST

  • A Binary Search Tree (BST) is a binary tree with following additional constraints:
  • For each node, the left subtree of a node must contain only nodes with keys less than the node's key and the right subtree of a node must contain only nodes with keys greater than the node's key.
  • In addition, there must be no duplicates

Find

Node* find (Node* node, int key)
{
    if (node == 0 || node->key == key)
    {   
        return node;
    }

    if (key > node->key)
        return find (node->right, key);
    else
        // key < node->key
        return find (node->left, key);
}

// given a node, find the child node with smallest value
Node* min_child_node (Node* node)
{
    Node* curr = node;

    while (curr && curr->left)
        curr = curr->left;

    return curr;
}

Node* del (Node* node, int key)
{
    if (node == 0) return 0;

    if (key < node->key)
        node->left = del (node->left, key);

    else if (key > node->key)
        node->right = del (node->right, key);

    else    // key == node->key
    {
        if (node->left == 0)
        {
            Node* temp = node->right;
            delete node;
            return temp;
        }
        else if (node->right == 0)
        {
            Node* temp = node->left;
            delete node;
            return temp;
        }
        else    // node with two children
        {
            Node* temp = min_child_node (node->right);

            // next value
            node->key = temp->key;

            node->right = del (node->right, temp->key);
        }
    }

    return node;
}
