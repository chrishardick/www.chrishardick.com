C++ List

  • Doubly linked list
  • constant time insertion and deletion
  • Like any list, lacks random access 
#include <list>

template < typename T>
class list;

Constructors

  • Default Constructor
    • Create an empty list 
    list<T> lst;
  • Copy Constructor
    • Copy supplied list 
    list lst(const list<T>& rhs);
  • Range [first,last) 
    list lst(InputIterator first, InputIterator last);

Destructor

  • Destruct all elements (destructor called for each) and destruct the container itself 
    ~list()

Iterator Invalidation Rules

  • Insersion - does not affect the validity of iterators
  • Deletion - only affects iterators the removed items

Member Functions

  • begin() 
    //
// returns iterator pointing to 1st element
// 
iterator begin();
const_iterator begin() const;
  • end() 
    //
// returns iterator pointing just past last element
//
iterator end();
const_iterator end() const;
  • front() 
    //
// return a reference to the 1st element in the container
//
// NOTES
// ===== 
// returns a reference (not an iterator)
// calling this function on an empty container is undefined
//
reference front();
const_reference front() const;
  • back() 
    //
// return a reference to the last element in the container
//
// NOTES
// ===== 
// returns a reference (not an iterator)
// calling this function on an empty container is undefined
//
reference back();
const_reference back() const;
  • empty() 
    //
// boolean - is list empty
// 
bool empty()
  • size() 
    //
// return number of elements in the container
//
size_type size() const;
  • push_back() 
    //
// add element to end of the container
//
// NOTE: throws exception if cannot allocate memory
//
void push_back(const T& t);
  • push_front() 
    //
// add element to beginning of the container
//
// NOTE: throws exception if cannot allocate memory
//
void push_front(const T& t);
  • pop_back() 
    //
// remove last element from the container
// - destructor will be called for the removed element
//
// NOTE: calling this on an empty container is undefined
//
void pop_back();
  • pop_front() 
    //
// remove 1st element from the container
// - destructor will be called for the removed element
//
// NOTE: calling this on an empty container is undefined
//
void pop_front();
  • insert() 
    //
// insert element into the list at the specified position
//
void insert(iterator position, const T& val);
  • erase() 
    //
// remove element(s) from the list
// - destructor will be called for the removed element(s)
// - for range version, the range goes from first to last - including first, but not including last
//
// RETURN
// ======
// iterator pointing to element which followed the last element erased. 
// will return end if last element was removed
//
iterator erase(iterator position);
iterator erase(iterator first, iterator last);      // remove [first, last)
  • clear() 
    //
// remove all elements from the list container
// - destructor will be called for each element removed
// 
void clear();
  • sort() 
    //
// sort the elements in the list (altering their position)
// 
void sort();    // use operator<()

void sort(Compare cmp); // Compare is a binary predicate taking two values of the same type as contained in the list

List as a Queue

  • A Queue is a linear data structure
  • FIFO (First In, First Out)
  • A list lends itself well as a base data structure for a queue as additions are at the rear and removals are from the front 
// using a list as a queue
    
#include <cassert>
    
std::list queue;

for (int i=0;i<5;i++)
{
    queue.push_back(i);
}

for (int i=0;i<5;i++)
{
    int x = queue.front();

    assert (x == i);

    queue.pop_front();
}
