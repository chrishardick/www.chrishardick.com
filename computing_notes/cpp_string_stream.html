C++ String Stream

#include <sstream>

ostringstream

  • An output string stream is used to write to a string.
  • Similar to snprintf() in C 
#include <sstream>

std::ostringstream oss;

int age = 22;
std::string name = "Angie";

// insertion operator
oss << name << " is " << age << "." << std::endl;

istringstream

  • An input string stream is used to read from a string.
  • Similar to sscanf() in C 
#include <sstream>

std::string age_str = "22";
std::istringstream iss (age_str);

int age;

// extraction operator
iss >> age;

std::cout << age << std::endl;      // 22

std::string str = "the extraction operator can be used to parse a string into words";

std::string word;

std::istringstream iss (str);

// extraction operator - extract word from a string
while (iss >> word)
{   
    std::cout >> word >> std::endl;
}
