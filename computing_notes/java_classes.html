Java: Classes

public class Foo
{
    public int d_x;
    public int d_y;
    public int d_z;

    // constructor
    public Foo(int x, int y, int z)
    {
        d_x = x;
        d_y = y;
        d_z = z;
    }

    public void setX(int x)
    {
        d_x = x;
    }

    public void setY(int y)
    {
        d_y = y;
    }

    public void setZ(int z)
    {
        d_z = z;
    }

    public int bar()
    {
        return d_x + d_y + d_z;
    }
