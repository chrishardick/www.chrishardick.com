Socket Address Structures

  • The basic purpose of socket address structures for IP based protocols is to specify an IP address and port
  • Socket address structures are passed to connect( ) -- to specify the IP address and port to connect to
  • Socket address structures are passed to bind( ) -- to specify the IP address and/or port to bind to
  • Socket address structures are returned from accept( ) -- to specify the IP address and/or port of the client
  • All socket address structures start w/ 'sockaddr_'
    • Generic socket address structure: sockaddr
    • IPv4 socket address structure: sockaddr_in
    • IPv6 socket address structure: sockaddr_in6

Generic Socket Address Structure: sockaddr

  • defined in <sys/socket.h>
  • Socket functions which accept socket address structures (bind, etc.) must accept all types of socket address structures (IPv4, IPv6, etc.)
  • The question arises as to which socket address structure type these functions should specify (the sockets API pre-dates the void pointer -- otherwise the void pointer could have been used)
  • The generic socket address structure fills this void
  • This requires all calls to these functions to cast the protocol specific socket address structure to a generic socket address structure
  • The generic socket structure serves as a template for the kernel. I.e. the kernel has to look at the sin_family member to determine the type of structure it was passed. Note that all the structures have sin_family located at the same offset. 
    /*
 * POSIX definition
 * - see POSIX Datatypes
 */
struct sockaddr
{
     uint8_t            sa_len;             // sizeof this struct 
     sa_family_t        sa_family;          // AF_INET|AF_INET6
     char               sa_data[14];
};


IPv4 Socket Address Structure: sockaddr_in

  • defined in <netinet/in.h>
  • POSIX only requires the following 3 fields: sin_family, sin_port & sin_addr 
    /*
 * POSIX definition
 * - see POSIX Datatypes
 */
struct sockaddr_in
{
     uint8_t            sin_len;       // length of struct (16 bytes)
     sa_family_t        sin_family;    // REQD BY POSIX: Address Family (AF_INET=IPv4)
     in_port_t          sin_port;      // REQD BY POSIX: 16 bit port# - network byte order
     struct in_addr     sin_addr;      // REQD BY POSIX: 32 bit IPv4 address - network byte order
     char               sin_zero[8];   // unused
};

struct in_addr
{
     /*
      * in_addr_t 
      * - POSIX datatype for IPv4 addresses
      * - Normally typedef'ed to uint32_t
      */
     in_addr_t     s_addr;       
};


IPv6 Socket Address Structure: sockaddr_in6

  • defined in <netinet/in.h> 
    struct sockaddr_in6
{
     uint8_t            sin6_len;           // sizeof this struct - 28 bytes
     sa_family_t        sin6_family;        // AF_INET6
     in_port_t          sin6_port;
     uint32_t           sin6_flowinfo;
     struct in6_addr    sin6_addr;          // 128 bit IPv6 address
     uint32_t           sin6_scope_id;
};

struct in6_addr
{
     uint8_t       s6_addr[16];        // IPv6 addresss (16 bytes - 128 bits)
};
