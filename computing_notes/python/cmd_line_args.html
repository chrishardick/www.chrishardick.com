Command Line Arguments & Options

Command Line Arguments

    import sys

# my_test.py
#
# if run as: my_test.py this is a test

# sys.argv is ['my_test.py', 'this', 'is', 'a', 'test']
print "sys.argv is ", sys.argv

# num args is 5
print "num args is ", len(sys.argv)
  • sys.argv is the list of command line arguments
  • len(sys.argv) is the # of command line arguments (including the executing script)
  • sys.argv[0] is the name of the executing script

Command Line Options

    # usage: my_test2.py [-h] [-i input]

import sys, getopt

in_file = ""

# opts - list of (option, value) pairs where option will be user specified option (prepended with '-')
# args - remaining command line arguments
opts, args = getopt.getopt(argv,"hi:")

for opt,arg in opts:
    if opt == '-h':
        usage()
        sys.exit()

    elif opt == '-i':
        in_file = arg

print "in_file: ", in_file
print "remaining args: ", args
