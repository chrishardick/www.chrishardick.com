Classes

  • All members are public
  • Per class variables are declared within the class definition and are referenced only by using the classname prefix (e.g. Class.num_instances in the example below)
  • Per instance variables are declared within the __init__ method
  • Unlike C++, you must use self to reference an object methods and data members from its methods
  • Unused objects are cleaned up via garbage collection. Objects are cleaned up when their reference count goes to 0 (zero) 
    # class definition

class Class():
    '''a simple class'''

    # class variables go here
    
    num_instances = 0
    num_foo_calls = 0

    #==========
    # __init__()
    # 
    # - constructor
    # - Python does not support multiple constructors, but you can define default parameters
    #==========
    def __init__(self,val=0):         # __init__() is the constructor

        # instance variables go here

        print "Class constructor called. val=", val

        self.i = val

        Class.num_instances += 1

    def bar(self):

        print "bar!"


    def foo(self):

        self.i += 1

        self.bar()

# class instantiation

a = Class(100)
b = Class(200)
c = Class(300)

print "a.i=", a.i
print "b.i=", b.i
print "c.i=", c.i

a.foo()

print "a.i=", a.i

print "# class instances: ", Class.num_instances
print "# foo calls:, Class.num_foo_calls
