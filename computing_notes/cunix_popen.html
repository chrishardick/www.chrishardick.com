C/UNIX: popen()

  • The popen() function creates a pipe, forks, and invokes the shell.
  • The usual use case is to executes an external command capturing the output 
#include <stdio.h>

//
// RETURN
// ======
// file descriptor which must be closed with pclose() rather than fclose() 
//
FILE* popen(const char* command         // command - c string
            ,const char* type           // "r" - reading; "w" - writing
            );


//
// pclose() waits for the associated process to terminate and returns the exit status of the command as returned by wait4()
//
int pclose(FILE* stream);


//
// typical usage
//

std::string cmd = "echo 'Hello, World!'";

FILE* file = popen(cmd.c_str(), "r");

if (file == 0)
    // error handling

char buf[1024];

std::string output;

while (fgets (buf, sizeof(buf), file))
    output.append(buf);

pclose(file);
