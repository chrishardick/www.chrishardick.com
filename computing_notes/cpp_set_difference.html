C++ - std::set_difference

#include <algorithm>


//
// perform set difference
// - i.e. items in first set, but not in second set
// - [first1,last1) - [first2,last2) 
// - note: inputs must be sorted
//
template <typename InputIterator1, typename InputIterator2, class OutputIterator>
OutputIterator< set_difference (InputIterator1 first1, InputIterator last1, 
                                    InputIterator2 first2, InputIterator last2,
                                    OutputIterator result);

// e.g.

std::vector<int> v1 { 20, 15,  5, 10, 25, 30 };
std::vector<int> v2 { 30, 20, 10 };

std::vector<int> v3;

std::sort (v1.begin(), v1.end());
std::sort (v2.begin(), v2.end());

std::set_difference (v1.begin(), v1.end()
                    ,v2.begin(), v2.end()
                    ,std::back_inserter(v3));

// 5 15 25
for (auto i = v3.begin(); i != v3.end(); i++)
{
    std::cout << *i << " ";
}

std::cout << std::endl;
