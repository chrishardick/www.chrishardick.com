Python: Requests

Requests - a Python library for HTTP

Quick Start

import requests

#
# get request
# - get a HTML document
# - r = Response object
#
r = requests.get('https://www.foo.com/api', qstr = {'key1':'value1', 'key2':'value2',...})

#
# post request
#
r = requests.post('https://www.foo.com/api', data = {'key1':'value1', 'key2':'value2',...})


# response - as text
r.text


# response as byte string (binary response)
r.content 


# json response
try:
    r.json()

except:

    // HANDLE JSON DECODING ERROR
