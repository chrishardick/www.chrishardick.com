Iterators

#include <iostream>
#include <string>
#include <cctype> // for toupper()

std::string str ("Hello, World!");

// const iterator
for (std::string::const_iterator it = str.begin(); it != str.end(); it++)
{
    std::cout << *it;
}

// non const iterator
for (std::string::iterator it = str.begin(); it != str.end(); it++)
{
    *it = toupper(*it);
}

getline()

//
// extract characters from the given input stream
// and append them to the given string object
// until the specified delimiter is found (or eof or an error)
// if the delimiter is found, it is extracted and discarded
//
// note(s)
// =======
// - the content of str will be replaced
//
// return
// ======
// the same istream reference as is 
//
std::istream& getline ( std::istream& is     // in/out
                        ,std::string& str    // out
                        ,char delim = '\n'   // in
                        );


//
// an example
//
std::string line;

getline (std::cin, line);

std::cout << "line: |" << line << "|" << std::endl;
