typedef basic_string<char> string
  • The string class handles bytes independent of the encoding used. E.g. with UTF-8 encoding, the length of the string (e.g. via the size() member function) may be greater than the number of unicode characters represented

Iterators

#include <iostream>
#include <string>
#include <cctype> // for toupper()

std::string str ("Hello, World!");

// const iterator
for (std::string::const_iterator it = str.begin(); it != str.end(); it++)
{
    std::cout << *it;
}

// non const iterator
for (std::string::iterator it = str.begin(); it != str.end(); it++)
{
    *it = toupper(*it);
}

//
// extract characters from the given input stream

                        ,std::string& str    // out
                        ,char delim = '\n'   // in
                        );


//
// an example
//
#include <sstream>

std::string line;


// traverse lines

while (std::getline(std::cin, line))
{
    std::istringstream iss (line);
    std::string fld;

    while (std::getline (iss, fld, '|'))
    {
        std::cout << "|" << fld << "|" << std::endl;
    }

    std::cout << std::endl;
}

Trim Whitespace

std::string rtrim (const std::string& s)
{
    // find last non-whitespace character
    size_t end = s.find_last_not_of (" \t");

    if (end == std::string::npos)       // not found
        return std::string();

    return s.substr(0,end+1);
}
