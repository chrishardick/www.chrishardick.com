pqxx - C++ Client Library for PostgreSQL

  • liqpqxx builds on top of PostgreSQL's standard C API libpq 
#include <iostream>
#include <pqxx/pqxx>

try
{
    // connect to the db
    pqxx::connection c;

    // start a transaction
    pqxx::work w(c);

    // execute a query
    // - exec1 executes a query returning a single row of data
    pqxx::row r = w.exec1("SELECT count(*) FROM foo");

    // commit the transaction
    // - make its work final
    // - without this, the work will be rolled back when the transaction object is destroyed.
    // - until then, you call the transactions exec, query_value, and stream functions (and variants) to execute SQL statements
    // - you pass the SQL statements as simple strings
    // - most of the exec funcdtions return a pqxx::result object which acts like a container of rows pqxx::row
    // - each row is a container of fields - pqxx::field
    // - each field's data is stored internally as a text string - in a format specified by PostgreSQL. 
    // - you 
    w.commit()
