C/UNIX: nanosleep()

  • Suspends the execution of the calling thread (i.e. put it to sleep) until either the time specified in req has elapsed, a signal is fired, or an error occurs
  • If interrupted by a signal, errno will be set to EINTR and the remaining time is written to rem (assuming it was not 0) -- which can then be passed to nanosleep again. 
#include <time.h>
#include <errno.h>        // for errno - if needed 
#include <string.h>       // for strerror() - if needed

#include <iostream>       // if needed

struct timespec
{
    time_t      tv_sec;     // seconds
    long        tv_nsec;    // nanoseconds (10^-9) 
};

//
// RETURN
// ======
//  0 = success
// -1 = otherwise. check errno.
//
int nanosleep ( const struct timespec* req
                ,struct timespec* rem           // optional. pass 0 if n/a 
                );


//
// sample usage
// 
struct timespec ts;
ts.tv_sec = 0;
ts.tv_nsec = 200 * 1E6;     // 200 ms 

int rc = nanosleep (&ts, 0);

if (rc != 0)
{
    std::cout << "ERROR CALLING nanosleep. RC=" << errno, << " " << strerror(errno) << std::endl;
}
