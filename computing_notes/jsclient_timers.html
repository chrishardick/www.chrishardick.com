setTimeout()

  • a method of the Window object
  • schedules a single function run after a specified number of milliseconds elapses 
//
// schedule a single function run after a specified number of milliseconds elapse
// - returns a value which can be passed to clearTimeout() to cancel the scheduled execution
// - accepts a variable number of arguments. 
// - 1st argument is the handler
// - 2nd argument is the number of milliseconds
// - any additional paramters are passed to the indicated function -- not supported in IE (i.e. not portable)
// 
var rc = setTimeout(update_clock, 1000 /* ms */);

setInterval()

  • a method of the Window object
  • registers a function to be invoked repeatedly at a specified interval 
//
// schedule a single function to be invoked repeatedly at the specified interval (# ms)
// - returns a value which can be passed to clearInterval() to cancel all future scheduled executions
// - accepts a variable number of arguments. 
// - 1st argument is the handler
// - 2nd argument is the number of milliseconds
// - any additional paramters are passed to the indicated function -- not supported in IE (i.e. not portable)
// 
var rc = setInterval(update_clock, 1000 /* ms */);

// cancel all future executions of scheduled function
clearInterval(rc);
