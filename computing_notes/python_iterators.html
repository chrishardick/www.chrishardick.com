Python: Iterators

  • Iterators are objects which can be iterated upon
  • In Python, an iterator is an object which must define __iter__() and __next__()
  • The __iter__() method returns the object itself
  • The __next__() moethod returns the next item in the sequence. After reaching the end, subsequent calls raise a StopIteration exception 
// a class which is an iterator
class Iter:

    def __init__(self):

        self.lst = []

        self.n = 0

    def __iter__(self):

        self.n = 0

        return self

    def __next__(self):

        if self.n < len(self.lst):

            ret = self.lst[self.n]
            
            self.n += 1

            return ret

        raise StopIteration

    def append(self,x):

        self.lst.append(x)

it = Iter()

it.append(1); it.append(2); it.append(3)

for i in it:
    print (i)
