C - Variable Argument Lists

C allows for functions which accept variable parameter lists (functions called @ different times w/ different parameters). 
#include <stdarg.h>

va_list pvar;

void va_start (va_list pvar, parmN);
type va_arg   (va_list pvar, type);
void va_end   (va_list pvar);
