switch statement

  • Unlike C/C++, there is no implicit fallthrough. I.e. you do not need the break statement 
switch value
{
    case value1:
        statements

    case value2,
         value3:

        statements

    default:
        statements
}

//
// example
//

let foo_char = "z"

switch (foo_char)
{
    case "a":
        print ("ape")
    case "z":
        print ("zebra")
    default:
        print ("animal")
