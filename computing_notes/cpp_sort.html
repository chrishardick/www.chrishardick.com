C++ std::sort

#include <algorithm>

// default
void sort (RandomAccessIterator first, RandomAccessIterator last);

// custom
// - Compare is a binary function which accepts elements of the container
//   and returns a boolean which indicates whether the lhs argument is less than the rhs argument
void sort (RandomAccessIterator first, RandomAccessIterator last, Compare comp);

// Utility
template <typename T>
std::ostream& operator<< (std::ostream& os, const std::vector<T>& v)
{
    for (auto a = v.begin(); a != v.end(); a++)
    {   
        os << *a << " ";
    }   

    return os; 
}

// put zeros at end
bool my_cmp (int a, int b)
{
    if (b == 0)
    {
        if (a == 0)
        {
            return false; // not less than
        }

        return true;
    }

    return a < b;
}

std::sort (a.begin(), a.end(), my_cmp);
