C++ File Streams

#include <fstream>

// stream class to write to files
typedef basic_ofstream<char> ofstream;

// stream class to read from files
typedef basic_ifstream<char> ifstream

// stream class to read from and write to files
typedef basic_fstream<char> fstream;

ofstream

  • Stream class to write to a file
  • Derived from ostream 
#include <fstream>

std::string filename ("data.dat");

std::ofstream of (filename, std::ios::out | std::ios::binary);

if (!of)
{
    // error handling
}

of.write (buf, buf_size);

of.close();

if (!of.good())
{
    // error handling
}

Constructors

//
// constructs an ofstream object - not yet associated with a file
//
ofstream();  

//
// constructs an ofstream object initially associated with specified file. 
// - if the file cannot be opened, the stream's failbit is set
// - mode bits: 
//      - in: file opened for reading
//      - out: file opened for writing (always set for ofstream)
//      - binary: operations are performed in binary mode rather than text mode
//
ofstream (  const char* filename
            ,ios_base::openmode mode = ios_base::out        // in: bitmap
            );

ifstream

  • Stream class to read from a file
  • Derived from istream 
#include <fstream>

std::string filename ("data.dat");

std::ifstream if (filename, std::ios::in | std::ios::binary);

if (!if)
{
    // error handling
}

char buf[100];

while (if.read(buf,sizeof(buf)))
{
    if (if.gcount() != sizeof(buf))
    {
        // error handling
    }

    // data processing
}

if.close();
