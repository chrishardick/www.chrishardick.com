Apache HTTP Server

# ServerRoot: top of the directory tree under which the server's 
# configuration, error, and log files are kept
ServerRoot "/etc/httpd"
ServerRoot "/etc/apache2"

# Dynamic Shared Object (DSO) Support
LoadModules module_name modules/module_name.so

# include config files
# - relative to ServerRoot
Include conf.d/*.conf

# User/Group the server should run as
User apache
Group apache

ServerAdmin chardick@optonline.net

# DocumentRoot: directory out of which you will serve your documents
# By default, all requests are taken from this directory, but
# symbolic links and aiases may point to other locations.
DocumentRoot "/var/www/html"

<Directory />
    Options FollowSymLinks
    AllowOverride None
</Directory>

# this should be changed to whatever your DocumentRoot is set to
<Directory "/var/www/html">
    ...
</Directory>

ScriptAlias /cgi-bin/ "/var/www/cgi-bin/"

<Directory "/var/www/cgi-bin">
    AllowOverride None
    Options None
    Order allow,deny
    Allow from all
</Directory>
  • restart server: sudo service apache2 restart
