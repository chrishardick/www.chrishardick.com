  • create table TABLENAME (field1-name field1-type field1-modifiers, ... fieldN-name fieldN-type fieldN-modifiers;
    • create table Event (	id int not null primary key auto_increment,
			object_id int not null,
			start_dt date not null,
			time time not null,
			freq char(1),
			inst varchar(50) not null,
			end_dt date,
			last_update timestamp,
			unique index ByStart (start_dt, time, id),
			index ByEnd (end_dt, id));


create table invoice_item (	id int not null primary key auto_increment, 
				date_of_service date not null, 
				invoice_id int, 
				action_id int not null, 
				invoice_item_type int not null,      
				flat_rate DECIMAL(14,2), 
				hour_rate DECIMAL(14,2), 
				num_hrs int, 
				unique index ByActionID (action_id,id), 
				unique index ByDate (date_of_service,id));
