IP Addresses: ASCII Strings ↔ Binary Values

  • Convert IP addresses between
    • ASCII strings (dotted decimal for IPv4, hex strings for IPv6) -and-
    • network byte ordered binary values
  • OLD: inet_aton( ), inet_ntoa( ) - only handle IPv4
  • NEW: inet_pton( ), inet_ntop( ) - handle both IPv4 and IPv6


inet_aton( ) and inet_ntoa( )

  • Older functions - only support IPv4
  • n = network
  • a = ASCII
  • NOTE: these functions accept socket address structures 
    #include <arpa/inet.h>

/*
 * ASCII (dotted decimal) => 32 bit IPv4 address in network byte order
 *
 * RETURN
 * ======
 * 1 = string is valid
 * 0 = invalid string
 */
int inet_aton(const char* str,         // in:     dotted decimal
              struct in_addr* addr);   // in/out: 32 bit IPv4 address in network byte order


/*
 * 32 bit IPv4 address in network byte order => ASCII (dotted decimal)
 *
 * NOTES
 * =====
 * - not re-enterant. Return value stored in static memory
 * 
 * RETURN
 * ======
 * ptr to dotted-decimal string
 * 
 */
char* inet_ntoa(struct in_addr addr);  // in:  32 bit IPv4 address

/*
 * ASCII (dotted decimal) => 32 bit IPv4 address in network byte order
 * 
 * NOTES
 * =====
 * - deprecated - use inet_aton( ) instead
 */
in_addr_t inet_addr (const char* str);


inet_pton( ) and inet_ntop( )

  • Newer functions - support both IPv4 and IPv6
  • n = network
  • p = presentation (rather than 'a' for ascii)
  • NOTE: these functions accept pointers to the address member of a socket address structure 
    #include <arpa/inet.h>

/*
 * PRESENTATION => NETWORK
 *
 * RETURN
 * ======
 *  1 = success
 *  0 = invalid string
 * -1 = errror
 */
int inet_pton (int family,        // in:  AF_INET|AF_INET6
               const char* str,   // in:  ascii string
               void* addr);       // out: struct in_addr* (IPv4) or struct in6_addr* (IPv6)


/*
 * NETWORK => PRESENTATION
 *
 * RETURN
 * ======
 * pointer to str = success
 * NULL = error
 */

#include <netinet/in.h>

#define INET_ADDRSTRLEN      16        // IPv4 presentation string size (dotted decimal - "192.168.000.001")
#define INET6_ADDRSTRLEN     46        // IPv6 presentation string size (hex string) 

const char* inet_ntop (int family,        // in:  AF_INET|AF_INET6
                       const void* addr,  // in:  struct in_addr* (IPv4) or struct in6_addr* (IPv6)
                       char* str,         // out: ascii string
                       size_t len);       // in:  sizeof(str) - either INET_ADDRSTRLEN or INET6_ADDRSTRLEN
